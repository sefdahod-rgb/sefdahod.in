DROP POLICY IF EXISTS "Principal can manage subject settings" ON public.class_subject_settings; CREATE POLICY "Owner and Principal can manage subject settings" ON public.class_subject_settings FOR ALL TO authenticated USING ( EXISTS ( SELECT 1 FROM public.user_roles WHERE user_id = auth.uid() AND role = 'owner' ) OR EXISTS ( SELECT 1 FROM public.staff_profiles WHERE auth_user_id = auth.uid() AND role = 'principal' AND is_active = true ) ) WITH CHECK ( EXISTS ( SELECT 1 FROM public.user_roles WHERE user_id = auth.uid() AND role = 'owner' ) OR EXISTS ( SELECT 1 FROM public.staff_profiles WHERE auth_user_id = auth.uid() AND role = 'principal' AND is_active = true ) );
